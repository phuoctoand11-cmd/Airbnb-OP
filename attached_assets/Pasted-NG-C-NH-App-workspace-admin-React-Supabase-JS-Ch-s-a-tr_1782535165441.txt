LƯU (onClick nút Lưu) -> gọi RPC, TÊN THAM SỐ phải khớp tuyệt đối:
await supabase.rpc("admin_update_employee", {
  p_employee_id: <id nhân viên>,
  p_email: email,
  p_role_name: role,           // giá trị kỹ thuật: admin/manager/sales/cleaner/maintenance/accountant/collaborator
  p_full_name: fullName,
  p_department_id: departmentId,
  p_position_id: positionId,
  p_employment_type: employmentType,
  p_status: status,
  p_team_id: teamId || null,
  p_phone: phone || null,
  p_date_of_birth: dateOfBirth || null,
  p_gender: gender || null,
  p_address: address || null,
  p_emergency_contact: emergencyContact || null,
  p_start_date: startDate || null,
  p_end_date: endDate || null,
  p_salary_base: salaryBase === "" ? null : Number(salaryBase),
  p_notes: notes || null,
  p_avatar_url: avatarUrl || null,
  p_password: password ? password : null
});
